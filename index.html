// Lista de textos disponíveis na exposição const listaTextos = ["./texto1.png", "./texto2.png"]; let indiceAtual = 0; // Elementos da interface const overlay = document.getElementById("overlay-texto"); const imagemExibida = document.getElementById("imagem-exibida"); const btnFechar = document.getElementById("btn-fechar"); const btnAnterior = document.getElementById("btn-anterior"); const btnProximo = document.getElementById("btn-proximo"); const indicador = document.getElementById("indicador-pagina"); // Função para atualizar a imagem e o indicador de página function atualizarImagem(novoIndice) { indiceAtual = novoIndice; imagemExibida.style.opacity = "0.3"; setTimeout(() => { imagemExibida.src = listaTextos[indiceAtual]; if (indicador) { indicador.textContent = `\({indiceAtual + 1} /\){listaTextos.length}`; } imagemExibida.style.opacity = "1"; }, 100); } // Ao carregar a página pelo QR Code window.addEventListener("DOMContentLoaded", () => { const urlParams = new URLSearchParams(window.location.search); const targetParam = urlParams.get('target'); // Se o QR Code tiver ?target=2 abre o segundo texto, caso contrário abre o primeiro if (targetParam === '2') { atualizarImagem(1); } else { atualizarImagem(0); } // Exibe o painel imediatamente ao aceder overlay.style.display = "flex"; }); // Eventos dos botões de navegação lateral if (btnProximo) { btnProximo.addEventListener("click", () => { const proximoIndice = (indiceAtual + 1) % listaTextos.length; atualizarImagem(proximoIndice); }); } if (btnAnterior) { btnAnterior.addEventListener("click", () => { const anteriorIndice = (indiceAtual - 1 + listaTextos.length) % listaTextos.length; atualizarImagem(anteriorIndice); }); } // Evento do botão de fechar (X) if (btnFechar) { btnFechar.addEventListener("click", () => { overlay.style.display = "none"; }); }
